function seedData() { if (!fs.existsSync(DATA_DIR)) { fs.mkdirSync(DATA_DIR, { recursive: true }); } if (!fs.existsSync(DATA_PATH)) { fs.writeFileSync(DATA_PATH, JSON.stringify({ rooms: [] }, null, 2)); } } function readState() { seedData(); const raw = fs.readFileSync(DATA_PATH, 'utf8'); try { return JSON.parse(raw); } catch (error) { return { rooms: [] }; } } function writeState(state) { seedData(); return Array.from({ length: 6 }, () => alphabet[Math.floor(Math.random() * alphabet.length)]).join(''); } function sanitizeMember(member) { return { id: member.id, displayName: member.displayName, avatar: member.avatar, joinedAt: member.joinedAt, points: member.points, status: member.status, badges: member.badges || [] }; } function sanitizeRoom(room) { return { id: room.id, uniqueRoomName: room.uniqueRoomName, roomCode: room.roomCode, creatorId: room.creatorId, createdAt: room.createdAt, members: room.members.map(sanitizeMember), messages: room.messages, polls: room.polls || [], quizzes: room.quizzes || [], games: room.games || [], midnight: room.midnight || { phase: 'night', round: 1, roles: [] } }; } function findRoomByNameMatching(roomName) { const state = readState(); const normalized = normalizeRoomName(roomName); return state.rooms.find(room => room.uniqueRoomName.toLowerCase() === normalized.toLowerCase()); } function getRoomById(roomId) { const state = readState(); return state.rooms.find(room => room.id === roomId); } function shouldDenyRoomAccess(room, memberId) { if (!room || !memberId) { return true; state.rooms.push(room); writeState(state); res.status(201).json({ success: true, room: { id: room.id, roomName: room.uniqueRoomName, roomCode: room.roomCode, displayName: String(displayName).trim(), avatar, memberId }, message: 'ROOM CREATED ✨' }); }); app.post('/api/rooms/join', (req, res) => { const { roomName, roomCode, displayName, avatar } = req.body || {}; if (!roomName || !roomCode || !displayName || !avatar) { return res.status(400).json({ success: false, message: 'Room name, room code, display name, and avatar are required.' }); } const normalizedRoomName = normalizeRoomName(roomName); const state = readState(); const room = state.rooms.find(candidate => candidate.uniqueRoomName.toLowerCase() === normalizedRoomName.toLowerCase() && candidate.roomCode.toLowerCase() === String(roomCode).trim().toLowerCase() ); if (!room) { return res.status(401).json({ success: false, message: 'Hmm… that room name and code do not match 🌙' }); } const memberAlreadyPresent = room.members.some(member => member.displayName.toLowerCase() === String(displayName).trim().toLowerCase()); writeState(state); res.json({ success: true, room: { id: room.id, roomName: room.uniqueRoomName, roomCode: room.roomCode, displayName: String(displayName).trim(), avatar, memberId }, message: 'Welcome to your private room.' });
