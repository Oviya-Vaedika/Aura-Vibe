const express = require('express'); const fs = require('fs'); const path = require('path'); const { randomUUID } = require('crypto'); const app = express(); const PORT = process.env.PORT || 3000; const DATA_DIR = path.join(__dirname, 'data'); const DATA_PATH = path.join(DATA_DIR, 'db.json'); app.use(express.json({ limit: '1mb' })); app.use(express.static(path.join(__dirname, 'public'))); function ensureDataStore() { if (!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR, { recursive: true }); if (!fs.existsSync(DATA_PATH)) { fs.writeFileSync(DATA_PATH, JSON.stringify({ rooms: [] }, null, 2)); } } function readState() { ensureDataStore(); try { const raw = fs.readFileSync(DATA_PATH, 'utf8'); const parsed = JSON.parse(raw); return { rooms: Array.isArray(parsed.rooms) ? parsed.rooms : [] }; } catch { return { rooms: [] }; } } function writeState(state) { ensureDataStore(); fs.writeFileSync(DATA_PATH, JSON.stringify(state, null, 2)); } function normalizeRoomName(input) { return String(input || '').trim().replace(/\s+/g, ' '); } function generateRoomCode() { const alphabet = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; let code = ''; for (let i = 0; i < 6; i += 1) { code += alphabet[Math.floor(Math.random() * alphabet.length)]; } return code; } function createMember(displayName, avatar) { return { id: randomUUID(), displayName: String(displayName || '').trim(), avatar, points: 10, status: 'online', badges: ['✨ New Club Member'], joinedAt: new Date().toISOString() }; } function sanitizeRoom(room) { return { id: room.id, roomName: room.roomName, roomCode: room.roomCode, creatorId: room.creatorId, createdAt: room.createdAt, members: room.members || [], messages: room.messages || [], polls: room.polls || [], quizzes: room.quizzes || [], games: room.games || [], midnight: room.midnight || { phase: 'night', round: 1, roles: [] } }; } function createDefaultRoomState(creatorMember, roomName, roomCode) { return { id: randomUUID(), roomName, roomCode, creatorId: creatorMember.id, createdAt: new Date().toISOString(), members: [creatorMember], messages: [ { id: randomUUID(), senderId: creatorMember.id, displayName: creatorMember.displayName, avatar: creatorMember.avatar, message: 'Welcome to the room! The vibe is officially live ✨', timestamp: new Date().toISOString() } ], polls: [ { id: randomUUID(), question: 'What should we play tonight?', options: ['Mystery game', 'Quiz battle', 'Movie night', 'Cozy chat'], closesAt: new Date(Date.now() + 86400000).toISOString(), votes: [{ userId: creatorMember.id, optionIndex: 0 }] } ], quizzes: [ { id: randomUUID(), title: 'Moonlight Warmup', description: 'A little spark to get everyone going.', questions: [ { question: 'Which color is the room vibe most famous for?', options: ['Indigo', 'Red', 'Green', 'Orange'], correctIndex: 0, timer: 25, points: 10 } ] } ], games: [ { id: randomUUID(), type: 'WHO SAID THAT?', status: 'ready' }, { id: randomUUID(), type: 'MOST LIKELY TO', status: 'ready' }, { id: randomUUID(), type: 'FRIEND TRIVIA', status: 'ready' } ], midnight: { phase: 'night', round: 1, roles: [{ id: creatorMember.id, role: 'CITIZEN' }], history: [] } }; } app.get('/api/health', (req, res) => { res.json({ ok: true, app: 'AURA VIBE', timestamp: new Date().toISOString() }); }); app.post('/api/rooms/create', (req, res) => { const { roomName, displayName, avatar } = req.body || {}; if (!roomName || !displayName || !avatar) { return res.status(400).json({ success: false, message: 'Room name, display name, and avatar are required.' }); } const normalizedName = normalizeRoomName(roomName); if (!normalizedName || normalizedName.length < 3) { return res.status(400).json({ success: false, message: 'Room name must be at least 3 characters long.' }); } const state = readState(); const duplicate = state.rooms.some(room => room.roomName.toLowerCase() === normalizedName.toLowerCase()); if (duplicate) { return res.status(409).json({ success: false, message: 'That room name is already taken. Try another one ✨' }); } const member = createMember(displayName, avatar); const roomCode = generateRoomCode(); const room = createDefaultRoomState(member, normalizedName, roomCode); state.rooms.push(room); writeState(state); return res.status(201).json({ success: true, room: { id: room.id, roomName: room.roomName, roomCode: room.roomCode, displayName: member.displayName, avatar: member.avatar, memberId: member.id }, message: 'ROOM CREATED ✨' }); }); app.post('/api/rooms/join', (req, res) => { const { roomName, roomCode, displayName, avatar } = req.body || {}; if (!roomName || !roomCode || !displayName || !avatar) { return res.status(400).json({ success: false, message: 'Room name, room code, display name, and avatar are required.' }); } const normalizedRoomName = normalizeRoomName(roomName); const state = readState(); const room = state.rooms.find(candidate => candidate.roomName.toLowerCase() === normalizedRoomName.toLowerCase() && candidate.roomCode.toLowerCase() === String(roomCode).trim().toLowerCase() ); if (!room) { return res.status(401).json({ success: false, message: 'Hmm… that room name and code do not match 🌙' }); } const member = createMember(displayName, avatar); room.members.push(member); room.messages.push({ id: randomUUID(), senderId: member.id, displayName: member.displayName, avatar: member.avatar, message: 'just entered the room and brought a fresh vibe ✨', timestamp: new Date().toISOString() }); writeState(state); return res.json({ success: true, room: { id: room.id, roomName: room.roomName, roomCode: room.roomCode, displayName: member.displayName, avatar: member.avatar, memberId: member.id }, message: 'Welcome to the room.' }); }); app.get('/api/rooms/:roomId', (req, res) => { const { roomId } = req.params; const { memberId } = req.query; const state = readState(); const room = state.rooms.find(candidate => candidate.id === roomId); if (!room) { return res.status(404).json({ success: false, message: 'This room could not be found.' }); } const member = room.members.find(candidate => candidate.id === memberId); if (!member) { return res.status(403).json({ success: false, message: 'Access denied. This room is private.' }); } return res.json({ success: true, room: sanitizeRoom(room), member }); }); app.post('/api/rooms/:roomId/messages', (req, res) => { const { roomId } = req.params; const { memberId, message } = req.body || {}; if (!memberId || !message) { return res.status(400).json({ success: false, message: 'Message and member are required.' }); } const state = readState(); const room = state.rooms.find(candidate => candidate.id === roomId); if (!room) { return res.status(404).json({ success: false, message: 'This room could not be found.' }); } const member = room.members.find(candidate => candidate.id === memberId); if (!member) { return res.status(403).json({ success: false, message: 'Access denied. This room is private.' }); } const text = String(message).trim(); if (!text) { return res.status(400).json({ success: false, message: 'Message cannot be empty.' }); } const entry = { id: randomUUID(), senderId: member.id, displayName: member.displayName, avatar: member.avatar, message: text, timestamp: new Date().toISOString() }; room.messages.push(entry); writeState(state); return res.status(201).json({ success: true, message: entry }); }); app.get('*', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); app.listen(PORT, () => { console.log(`AURA VIBE is running on http://localhost:${PORT}`); });
